    <!-- Step A/B: Available Slots -->

    <div id="slot-section">

      <h2 class="text-lg font-semibold mb-2">Available Slots</h2>

        <p class="text-gray-500">Loading live availability...</p>

    <!-- Step C/D: Checkout Modal -->

    <div id="checkout-section" class="hidden mt-6 pt-6 border-t">

      <h2 class="text-lg font-semibold text-gray-800 mb-2">Complete Reservation</h2>

      <p id="selected-info" class="text-sm text-gray-600 mb-4"></p>

      

      <div class="bg-gray-50 p-4 rounded-lg mb-4 text-center">

        <p class="text-xs text-gray-500 uppercase font-bold">Scan to Pay via GCash</p>

        <!-- Your QR Code will display here -->

        <img id="gcash-qr-img" src="" alt="GCash QR Code" class="w-48 h-48 mx-auto my-2 rounded border" />

        <p id="fee-breakdown" class="text-lg font-bold text-green-800"></p>

      </div>


      <form id="booking-form" class="space-y-3">

        <div>

          <label class="block text-sm font-medium text-gray-700">Your Email</label>

          <input type="email" id="email" required class="w-full mt-1 p-2 border rounded-md" placeholder="name@example.com" />

        </div>

        <div>

          <label class="block text-sm font-medium text-gray-700">GCash Reference Number</label>

          <input type="text" id="gcash-ref" required class="w-full mt-1 p-2 border rounded-md" placeholder="13-digit reference no." />

        </div>

        <button type="submit" id="submit-btn" class="w-full bg-green-600 hover:bg-green-700 text-white font-bold py-2 rounded-md">

          Confirm Payment & Book

        </button>

  <script>

    // ==========================================

    // PASTE YOUR TWO LINKS HERE:

    // ==========================================

    const API_URL = "https://script.google.com/macros/s/AKfycbxjcGWN_iyGH26kuzwY_FX7qnHn365D3evpZ-cqZxYoRNO6IrmIgIJC2BVlwG6PiLIxXQ/exec"; 

    const QR_IMAGE_URL = "https://i.postimg.cc/cLdL90D5/IMG-3613.jpg"; 

    // ==========================================


    let selectedSlot = null;

    document.getElementById('gcash-qr-img').src = QR_IMAGE_URL;


    // Fetch live slots from Google Apps Script

    async function fetchSlots() {

      const container = document.getElementById('slots-container');

      try {

        const res = await fetch(API_URL);

        const slots = await res.json();

        

        if (slots.length === 0) {

          container.innerHTML = '<p class="text-gray-500">No available slots right now.</p>';

          return;

        }


        container.innerHTML = slots.map(slot => `

          <div onclick="selectSlot(${slot.SlotID}, '${slot.CourtName}', '${slot.TimeSlot}', ${slot.Price})" 

               class="p-3 border rounded-lg hover:border-green-500 cursor-pointer flex justify-between items-center transition">

            <div>

              <p class="font-bold text-gray-800">${slot.CourtName}</p>

              <p class="text-sm text-gray-500">${slot.TimeSlot}</p>

            </div>

            <span class="text-green-600 font-bold">₱${slot.Price}</span>

          </div>

        `).join('');

      } catch (err) {

        container.innerHTML = '<p class="text-red-500">Failed to load slots.</p>';

      }

    }


    // Handle slot selection

    function selectSlot(id, court, time, price) {

      selectedSlot = { id, court, time, price };

      document.getElementById('checkout-section').classList.remove('hidden');

      document.getElementById('selected-info').innerText = `${court} | ${time}`;

      document.getElementById('fee-breakdown').innerText = `Total: ₱${price}`;

    }


    // Submit booking payload

    document.getElementById('booking-form').addEventListener('submit', async (e) => {

      e.preventDefault();

      const btn = document.getElementById('submit-btn');

      const statusMsg = document.getElementById('status-message');

      

      btn.disabled = true;

      btn.innerText = 'Processing...';


      const payload = {

        slotId: selectedSlot.id,

        email: document.getElementById('email').value,

        gcashRef: document.getElementById('gcash-ref').value

      };


      try {

        const res = await fetch(API_URL, {

          method: 'POST',

          body: JSON.stringify(payload)

        });

        const result = await res.json();


        if (result.success) {

          statusMsg.className = 'mt-4 p-3 bg-green-100 text-green-800 rounded-md text-center';

          statusMsg.innerText = `Booking Submitted! Ref ID: ${result.bookingId}. Check your email.`;

          statusMsg.classList.remove('hidden');

          document.getElementById('booking-form').reset();

          document.getElementById('checkout-section').classList.add('hidden');

          fetchSlots(); 

        } else {

          throw new Error(result.message || 'Booking failed');

        }

      } catch (err) {

        statusMsg.className = 'mt-4 p-3 bg-red-100 text-red-800 rounded-md text-center';

        statusMsg.innerText = err.message;

        statusMsg.classList.remove('hidden');

      } finally {

        btn.disabled = false;

        btn.innerText = 'Confirm Payment & Book';

      }

    });


    fetchSlots();

  </script>
